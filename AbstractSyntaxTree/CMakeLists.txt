cmake_minimum_required(VERSION 3.20)

project(abstract_syntax_tree
  VERSION 1.0.0
  DESCRIPTION "Sample sources for exploring Clang's abstract syntax tree"
  LANGUAGES CXX
)

# ------------------------------------------------------------------------------
# Project-wide policy
# ------------------------------------------------------------------------------
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR
    "In-source builds are not permitted. Use: cmake -S . -B build")
endif()

set(ast_cxx_standard 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Emit compile_commands.json so that clangd, clang-tidy and LibTooling-based
# programs can consume the build configuration directly.
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

option(AST_ENABLE_DUMP_TARGETS
  "Generate Clang AST artifacts as part of the default build (requires Clang)" ON)

# ------------------------------------------------------------------------------
# Warnings (interface target, applied per consumer rather than globally)
# ------------------------------------------------------------------------------
add_library(ast_warnings INTERFACE)
target_compile_options(ast_warnings INTERFACE
  $<$<CXX_COMPILER_ID:MSVC>:/W4>
  $<$<NOT:$<CXX_COMPILER_ID:MSVC>>:-Wall -Wextra -Wpedantic>
)

# ------------------------------------------------------------------------------
# Library: f() lives in test.cc and is declared in test.h
# ------------------------------------------------------------------------------
set(ast_headers "${CMAKE_CURRENT_SOURCE_DIR}/test.h")

add_library(ast_sample STATIC
  test.cc
  ${ast_headers}
)
add_library(AbstractSyntaxTree::ast_sample ALIAS ast_sample)

target_include_directories(ast_sample PUBLIC
  "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>"
)
target_compile_features(ast_sample PUBLIC cxx_std_${ast_cxx_standard})
target_link_libraries(ast_sample PRIVATE ast_warnings)

# ------------------------------------------------------------------------------
# Executable: calls f() from the library
# ------------------------------------------------------------------------------
add_executable(ast_demo main.cc)
target_link_libraries(ast_demo PRIVATE ast_sample ast_warnings)

# ------------------------------------------------------------------------------
# AST artifacts, generated by default (part of the "all" target)
#
# For every translation unit of the targets registered below, the build writes
# the following files under <build>/ast/<target>/ :
#
#   <stem>.ast        serialized binary AST   (clang -emit-ast)
#   <stem>.ast.txt    human-readable dump     (-Xclang -ast-dump)
#   <stem>.ast.json   machine-readable dump   (-Xclang -ast-dump=json)
#
# The sources are read from each target's SOURCES property, so a source added
# to a registered target is picked up automatically. Without Clang the
# artifacts are skipped and the regular build is unaffected. To disable them:
#
#   cmake -S . -B build -DAST_ENABLE_DUMP_TARGETS=OFF
# ------------------------------------------------------------------------------
if(AST_ENABLE_DUMP_TARGETS)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    set(ast_clang_executable "${CMAKE_CXX_COMPILER}")
  else()
    find_program(ast_clang_executable NAMES clang++ clang)
  endif()

  if(ast_clang_executable)
    set(ast_output_dir "${CMAKE_CURRENT_BINARY_DIR}/ast")
    set(ast_dump_script "${CMAKE_CURRENT_SOURCE_DIR}/cmake/dump_ast.cmake")

    # Umbrella target; each registered target attaches its own dependency to it.
    add_custom_target(ast_dump ALL
      COMMENT "Clang AST artifacts are up to date")

    # ast_add_artifacts(<target> [FILTER <substring>])
    #
    # FILTER restricts the textual and JSON dumps to declarations whose
    # qualified name contains <substring>, which keeps the output small for
    # sources that include large standard-library headers.
    function(ast_add_artifacts target)
      cmake_parse_arguments(PARSE_ARGV 1 arg "" "FILTER" "")

      get_target_property(target_sources ${target} SOURCES)
      set(output_dir "${ast_output_dir}/${target}")
      set(artifact_files "")

      foreach(source IN LISTS target_sources)
        if(NOT source MATCHES "\\.(cc|cpp|cxx)$")
          continue() # headers are tracked through ast_headers instead
        endif()

        cmake_path(ABSOLUTE_PATH source
          BASE_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
          NORMALIZE
          OUTPUT_VARIABLE source_file)
        cmake_path(GET source_file STEM LAST_ONLY stem)

        # Serialized AST: Clang writes the file itself, so no redirection is needed.
        set(binary_file "${output_dir}/${stem}.ast")
        add_custom_command(
          OUTPUT "${binary_file}"
          COMMAND "${CMAKE_COMMAND}" -E make_directory "${output_dir}"
          COMMAND "${ast_clang_executable}" -std=c++${ast_cxx_standard}
          -I "${CMAKE_CURRENT_SOURCE_DIR}"
          -emit-ast "${source_file}" -o "${binary_file}"
          DEPENDS "${source_file}" ${ast_headers}
          COMMENT "Serializing AST: ${target}/${stem}.ast"
          VERBATIM
        )
        list(APPEND artifact_files "${binary_file}")

        # Textual and JSON dumps: routed through a script, because
        # add_custom_command() cannot portably redirect standard output.
        foreach(format extension IN ZIP_LISTS ast_dump_formats ast_dump_extensions)
          set(dump_file "${output_dir}/${stem}.ast.${extension}")
          add_custom_command(
            OUTPUT "${dump_file}"
            COMMAND "${CMAKE_COMMAND}" -E make_directory "${output_dir}"
            COMMAND "${CMAKE_COMMAND}"
            -D "clang=${ast_clang_executable}"
            -D "standard=${ast_cxx_standard}"
            -D "format=${format}"
            -D "source=${source_file}"
            -D "output=${dump_file}"
            -D "include_dir=${CMAKE_CURRENT_SOURCE_DIR}"
            -D "filter=${arg_FILTER}"
            -P "${ast_dump_script}"
            DEPENDS "${source_file}" ${ast_headers} "${ast_dump_script}"
            COMMENT "Dumping AST (${format}): ${target}/${stem}.ast.${extension}"
            VERBATIM
          )
          list(APPEND artifact_files "${dump_file}")
        endforeach()
      endforeach()

      add_custom_target(ast_dump_${target} DEPENDS ${artifact_files})
      add_dependencies(ast_dump ast_dump_${target})
    endfunction()

    set(ast_dump_formats default json)
    set(ast_dump_extensions txt json)

    ast_add_artifacts(ast_sample)
    ast_add_artifacts(ast_demo FILTER main)
  else()
    message(STATUS "Clang not found; AST artifacts will not be generated")
  endif()
endif()
